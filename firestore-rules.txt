rules_version = '2';

service cloud.firestore {
  match /databases/{database}/documents {
    function isSignedIn() {
      return request.auth != null;
    }

    function isAdmin() {
      return isSignedIn() && request.auth.token.email == "jlucaspaisig@gmail.com";
    }

    function ownsConversation(conversationId) {
      return isSignedIn()
        && exists(/databases/$(database)/documents/loveConversations/$(conversationId))
        && get(/databases/$(database)/documents/loveConversations/$(conversationId)).data.visitorUid == request.auth.uid;
    }

    match /loveConversations/{conversationId} {
      allow create: if isSignedIn()
        && request.resource.data.visitorUid == request.auth.uid;

      allow read, update: if isAdmin()
        || (isSignedIn() && resource.data.visitorUid == request.auth.uid);

      match /messages/{messageId} {
        allow read: if isAdmin() || ownsConversation(conversationId);

        allow create: if isAdmin()
          || (ownsConversation(conversationId)
            && request.resource.data.sender == "visitor");
      }
    }
  }
}
